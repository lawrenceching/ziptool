### Task 4: Release workflow

**Files:**
- Create: `.github/workflows/release.yml`
- Modify: `_docs/release.md` §5(artifact 名从 `bundle-<triple>` 改为 `bundle-${{ matrix.os }}`,OS label 已足够区分,免去跨 step 传递 triple 的机制)

- [ ] **Step 1: 创建 `.github/workflows/release.yml`,内容如下(完整)**

```yaml
name: Release

permissions:
  contents: read

on:
  workflow_dispatch:
    inputs:
      tag:
        description: 要发布的 tag (如 v0.1.0)
        required: true
        type: string

concurrency:
  group: release-${{ inputs.tag }}
  cancel-in-progress: true

jobs:
  verify:
    name: verify (tag/版本一致性 + CI 门禁)
    runs-on: ubuntu-latest
    timeout-minutes: 115
    env:
      GH_TOKEN: ${{ github.token }}
      GH_REPO: ${{ github.repository }}
      TAG: ${{ inputs.tag }}
    steps:
      - uses: actions/checkout@v4
        with:
          ref: ${{ inputs.tag }}
      - name: 校验 tag 与版本一致
        run: |
          VER="${TAG#v}"
          CARGO_VER="$(grep -m1 '^version =' Cargo.toml | cut -d '"' -f2)"
          TAURI_VER="$(jq -r .version crates/tauri-app/tauri.conf.json)"
          echo "tag=$TAG ver=$VER cargo=$CARGO_VER tauri=$TAURI_VER"
          if [ "$VER" != "$CARGO_VER" ] || [ "$VER" != "$TAURI_VER" ]; then
            echo "::error::版本不一致: tag=$VER Cargo.toml=$CARGO_VER tauri.conf=$TAURI_VER"
            exit 1
          fi
      - name: 等待该 commit 的 CI 全绿
        run: |
          SHA="$(git rev-parse HEAD)"
          echo "等待 CI: commit $SHA"
          deadline_wait=$(( $(date +%s) + 600 ))    # run 出现上限 10 分钟
          deadline_done=$(( $(date +%s) + 6000 ))   # 完成上限 100 分钟
          appeared=0
          while :; do
            now=$(date +%s)
            status="$(gh run list --workflow=CI --commit="$SHA" --json status --limit 20 | jq -r '.[0].status // "none"')"
            conclusion="$(gh run list --workflow=CI --commit="$SHA" --json conclusion --limit 20 | jq -r '.[0].conclusion // "none"')"
            [ "$status" != "none" ] && appeared=1
            if [ "$appeared" = 1 ]; then
              if [ "$status" = "completed" ]; then
                if [ "$conclusion" = "success" ]; then
                  echo "CI 全绿: $SHA"
                  exit 0
                fi
                echo "::error::CI 结论为 $conclusion: $SHA"
                exit 1
              fi
              if [ "$now" -ge "$deadline_done" ]; then
                echo "::error::等待 CI 完成超时 (100 分钟)"
                exit 1
              fi
            elif [ "$now" -ge "$deadline_wait" ]; then
              echo "::error::10 分钟内未发现该 commit 的 CI run (该 commit 是否已推送?)"
              exit 1
            fi
            echo "CI 状态: status=$status conclusion=$conclusion, 30s 后重查"
            sleep 30
          done

  build:
    name: build (${{ matrix.os }})
    needs: verify
    runs-on: ${{ matrix.os }}
    timeout-minutes: 60
    strategy:
      fail-fast: false
      matrix:
        os: [ubuntu-latest, ubuntu-24.04-arm, windows-latest, windows-11-arm, macos-latest]
    env:
      TAG: ${{ inputs.tag }}
    steps:
      - uses: actions/checkout@v4
        with:
          ref: ${{ inputs.tag }}
      - uses: dtolnay/rust-toolchain@stable
      - uses: Swatinem/rust-cache@v2
        with:
          cache-on-failure: true
      - name: Install Tauri dependencies (Linux)
        if: runner.os == 'Linux'
        run: |
          sudo apt-get update
          sudo apt-get install -y libwebkit2gtk-4.1-dev libgtk-3-dev libayatana-appindicator3-dev librsvg2-dev libxdo-dev libssl-dev
      - name: Build CLI
        run: cargo build --release -p zip-tool-cli
      - name: Package CLI (unix)
        if: runner.os != 'Windows'
        run: |
          VER="${TAG#v}"
          TRIPLE="$(rustc -vV | sed -n 's/^host: //p')"
          mkdir -p dist
          tar czf "dist/ziptool-cli-${VER}-${TRIPLE}.tar.gz" -C target/release ziptool
      - name: Package CLI (Windows)
        if: runner.os == 'Windows'
        shell: pwsh
        run: |
          $ver = $env:TAG.TrimStart('v')
          $triple = (rustc -vV | Select-String '^host: ').Line -replace '^host: ', ''
          New-Item -ItemType Directory -Force dist | Out-Null
          Compress-Archive -Path target/release/ziptool.exe -DestinationPath "dist/ziptool-cli-$ver-$triple.zip"
      - uses: actions/setup-node@v4
        with:
          node-version: 22
          cache: npm
          cache-dependency-path: crates/tauri-app/package-lock.json
      - name: Tauri build
        working-directory: crates/tauri-app
        run: npm ci && npx tauri build
      - name: Collect bundles (unix)
        if: runner.os != 'Windows'
        run: |
          b=target/release/bundle
          cp "$b"/deb/*.deb dist/ 2>/dev/null || true
          cp "$b"/rpm/*.rpm dist/ 2>/dev/null || true
          cp "$b"/appimage/*.AppImage dist/ 2>/dev/null || true
          cp "$b"/dmg/*.dmg dist/ 2>/dev/null || true
          ls -l dist
      - name: Collect bundles (Windows)
        if: runner.os == 'Windows'
        shell: pwsh
        run: |
          Copy-Item target/release/bundle/nsis/*-setup.exe dist/
          Copy-Item target/release/bundle/msi/*.msi dist/
          Get-ChildItem dist
      - uses: actions/upload-artifact@v4
        with:
          name: bundle-${{ matrix.os }}
          path: dist

  release:
    name: release
    needs: build
    runs-on: ubuntu-latest
    timeout-minutes: 10
    permissions:
      contents: write
    env:
      GH_TOKEN: ${{ github.token }}
      GH_REPO: ${{ github.repository }}
      TAG: ${{ inputs.tag }}
    steps:
      - uses: actions/download-artifact@v4
        with:
          pattern: bundle-*
          merge-multiple: true
          path: dist
      - name: Create release
        run: |
          ls -l dist
          if gh release view "$TAG" --json id >/dev/null 2>&1; then
            gh release upload "$TAG" dist/* --clobber
          else
            gh release create "$TAG" dist/* --title "ziptool $TAG" --generate-notes --verify-tag
          fi
```
